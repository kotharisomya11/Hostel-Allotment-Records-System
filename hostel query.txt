
-- A. CREATION AND USE

CREATE DATABASE HOSTELS;
USE HOSTELS;

-- ============================================================
-- B. CREATION OF TABLES
-- ============================================================

-- 1. STUDENT
CREATE TABLE STUDENT(
  ROLL_NO VARCHAR(15) NOT NULL,
  FIRST_NAME VARCHAR(25) NOT NULL,
  LAST_NAME VARCHAR(25),
  ROOM_NO INT(4),
  MAIL_ID VARCHAR(50),
  DOB DATE,
  AGE INT(3),
  CONTACT_NO BIGINT(13),
  ADD_HOUSE_NO INT(3),
  ADD_PIN INT(8) NOT NULL,
  ADD_STREET VARCHAR(15),
  PRIMARY KEY(ROLL_NO)
);

-- 2. ADDRESS
CREATE TABLE ADDRESS(
  ADD_PIN INT(8) NOT NULL,
  ADD_CITY VARCHAR(30),
  ADD_STATE VARCHAR(40),
  ADD_COUNTRY VARCHAR(20),
  PRIMARY KEY(ADD_PIN)
);
DESCRIBE ADDRESS;

-- 3. HOSTEL
CREATE TABLE HOSTEL(
  H_NAME VARCHAR(15) NOT NULL,
  CAPACITY INT(5) NOT NULL,
  HOSTEL_NO INT(2) NOT NULL,
  F_SPORTS VARCHAR(15),
  F_NEWSPAP_NAME VARCHAR(20),
  F_MESS_id VARCHAR(5),
  VACANT_ROOMS INT(3),
  PRIMARY KEY(HOSTEL_NO)
);
DESCRIBE HOSTEL;

-- 4. SKILLED EMPLOYEE
CREATE TABLE S_EMPLOYEE(
  ID_NO VARCHAR(5) NOT NULL,
  F_NAME VARCHAR(20) NOT NULL,
  L_NAME CHAR(20),
  SEX CHAR(6),
  AGE INT(3),
  PH_NO BIGINT(12),
  POST VARCHAR(15),
  MAIL_ID VARCHAR(50),
  SALARY INT(6),
  PRIMARY KEY(ID_NO)
);
DESCRIBE S_EMPLOYEE;

-- 5. UNSKILLED EMPLOYEE
CREATE TABLE U_EMPLOYEE(
  ID_NO VARCHAR(5) NOT NULL,
  F_NAME VARCHAR(20) NOT NULL,
  L_NAME CHAR(20),
  SEX CHAR(6),
  AGE INT(3),
  PH_NO BIGINT(12),
  WORK_AS VARCHAR(15),
  CONTACTOR VARCHAR(50),
  WAGES INT(6),
  PRIMARY KEY(ID_NO)
);
DESCRIBE U_EMPLOYEE;

-- 6. STUDENT_IN_OUT
CREATE TABLE STUDENT_IN_OUT(
  RECORD_NO INT(5),
  ROLL_NO VARCHAR(15) NOT NULL,
  STUDENT_ARR TIMESTAMP NULL,
  STUDENT_DEP TIMESTAMP NULL,
  PRIMARY KEY(RECORD_NO)
);
DESCRIBE STUDENT_IN_OUT;

-- 7. VISITOR
CREATE TABLE VISITOR(
  RECORD_NO INT(5),
  F_NAME VARCHAR(15) NOT NULL,
  L_NAME VARCHAR(15),
  VISITOR_ARR TIMESTAMP NULL,
  VISITOR_DEP TIMESTAMP NULL,
  VISITOR_CONTACT BIGINT(12),
  HOSTEL_NO INT(2),
  PRIMARY KEY(RECORD_NO)
);
DESCRIBE VISITOR;

-- 8. SKILLED EMPLOYEE TRANSACTION
CREATE TABLE S_EMPLOYEE_TRAN(
  ID_NO VARCHAR(5),
  ATTENDENCE VARCHAR(2),
  HOSTEL_NO INT(2),
  WORKING_DAYS INT(2),
  PRIMARY KEY(ID_NO)
);
DESCRIBE S_EMPLOYEE_TRAN;

-- 9. UNSKILLED EMPLOYEE TRANSACTION
CREATE TABLE U_EMPLOYEE_TRAN(
  ID_NO VARCHAR(5),
  ATTENDENCE VARCHAR(2),
  HOSTEL_NO INT(2),
  WORKING_DAYS INT(2),
  PRIMARY KEY(ID_NO)
);
DESCRIBE U_EMPLOYEE_TRAN;

-- 10. STUDENT TRANSACTION
CREATE TABLE STUDENT_TRAN(
  ROLL_NO VARCHAR(15),
  ATTENDENCE VARCHAR(2),
  HOSTEL_NO INT(2),
  LIVING_DAYS INT(2),
  PRIMARY KEY(ROLL_NO)
);
DESCRIBE STUDENT_TRAN;

-- 11. INTERNET
CREATE TABLE INTERNET(
  HOSTEL_NO INT(2),
  ROOM_NO INT(4),
  IP_ADDRESS VARCHAR(16),
  PRIMARY KEY(HOSTEL_NO, ROOM_NO)
);
DESCRIBE INTERNET;

-- 12. MESS_ADMIN
CREATE TABLE MESS_ADMIN(
  MESS_ID VARCHAR(5),
  CATARER VARCHAR(15),
  REPRESENTATIVE VARCHAR(15),
  PRIMARY KEY(MESS_ID)
);
DESCRIBE MESS_ADMIN;

-- 13. MESS_MENU
CREATE TABLE MESS_MENU(
  MESS_ID VARCHAR(5),
  M_DAY VARCHAR(10),
  M_TIME VARCHAR(10),
  MENU VARCHAR(80),
  PRIMARY KEY(MESS_ID, M_DAY, M_TIME)
);
DESCRIBE MESS_MENU;

-- ============================================================
-- C. ALTERING AND REFERENCING (FOREIGN KEYS)
-- ============================================================

-- 1. VISITOR -> HOSTEL (HOSTEL_NO)
ALTER TABLE HOSTELS.VISITOR ADD CONSTRAINT FK_VISITOR_HOSTEL FOREIGN KEY (HOSTEL_NO)
  REFERENCES HOSTELS.HOSTEL(HOSTEL_NO) ON UPDATE CASCADE ON DELETE CASCADE;

-- 2. STUDENT -> ADDRESS (ADD_PIN)
ALTER TABLE HOSTELS.STUDENT ADD CONSTRAINT FK_STUDENT_PIN FOREIGN KEY (ADD_PIN)
  REFERENCES HOSTELS.ADDRESS(ADD_PIN) ON UPDATE CASCADE ON DELETE CASCADE;

-- 3. HOSTEL -> MESS_ADMIN (MESS_ID)
ALTER TABLE HOSTELS.HOSTEL ADD CONSTRAINT FK_HOSTEL_MESS FOREIGN KEY (F_MESS_id)
  REFERENCES HOSTELS.MESS_ADMIN(MESS_ID) ON UPDATE CASCADE ON DELETE CASCADE;

-- 4. MESS_MENU -> MESS_ADMIN (MESS_ID)
ALTER TABLE HOSTELS.MESS_MENU ADD CONSTRAINT FK_MESSMENU_MESSID FOREIGN KEY (MESS_ID)
  REFERENCES HOSTELS.MESS_ADMIN(MESS_ID) ON UPDATE CASCADE ON DELETE CASCADE;

-- 5. STUDENT_IN_OUT -> STUDENT (ROLL_NO)
ALTER TABLE HOSTELS.STUDENT_IN_OUT ADD CONSTRAINT FK_SIO_ROLLNO FOREIGN KEY (ROLL_NO)
  REFERENCES HOSTELS.STUDENT(ROLL_NO) ON UPDATE CASCADE ON DELETE CASCADE;

-- 6. S_EMPLOYEE_TRAN -> S_EMPLOYEE, HOSTEL
ALTER TABLE HOSTELS.S_EMPLOYEE_TRAN
  ADD CONSTRAINT FK_SETRAN_ID FOREIGN KEY (ID_NO)
    REFERENCES HOSTELS.S_EMPLOYEE(ID_NO) ON UPDATE CASCADE ON DELETE CASCADE,
  ADD CONSTRAINT FK_SETRAN_HOSTEL FOREIGN KEY (HOSTEL_NO)
    REFERENCES HOSTELS.HOSTEL(HOSTEL_NO) ON UPDATE CASCADE ON DELETE CASCADE;

-- 7. U_EMPLOYEE_TRAN -> U_EMPLOYEE, HOSTEL
ALTER TABLE HOSTELS.U_EMPLOYEE_TRAN
  ADD CONSTRAINT FK_UETRAN_ID FOREIGN KEY (ID_NO)
    REFERENCES HOSTELS.U_EMPLOYEE(ID_NO) ON UPDATE CASCADE ON DELETE CASCADE,
  ADD CONSTRAINT FK_UETRAN_HOSTEL FOREIGN KEY (HOSTEL_NO)
    REFERENCES HOSTELS.HOSTEL(HOSTEL_NO) ON UPDATE CASCADE ON DELETE CASCADE;

-- 8. INTERNET -> HOSTEL (HOSTEL_NO)
ALTER TABLE HOSTELS.INTERNET ADD CONSTRAINT FK_INTERNET_HOSTEL FOREIGN KEY (HOSTEL_NO)
  REFERENCES HOSTELS.HOSTEL(HOSTEL_NO) ON UPDATE CASCADE ON DELETE CASCADE;

-- 9. STUDENT_TRAN -> STUDENT, HOSTEL
ALTER TABLE HOSTELS.STUDENT_TRAN
  ADD CONSTRAINT FK_STRAN_ROLLNO FOREIGN KEY (ROLL_NO)
    REFERENCES HOSTELS.STUDENT(ROLL_NO) ON UPDATE CASCADE ON DELETE CASCADE,
  ADD CONSTRAINT FK_STRAN_HOSTEL FOREIGN KEY (HOSTEL_NO)
    REFERENCES HOSTELS.HOSTEL(HOSTEL_NO) ON UPDATE CASCADE ON DELETE CASCADE;

-- 1. Student info by room number
SELECT * FROM STUDENT WHERE ROOM_NO = 145;

-- 2. Number of vacant rooms in a hostel (column is VACANT_ROOMS, not vacant_room)
SELECT VACANT_ROOMS FROM HOSTEL WHERE HOSTEL_NO = 1;

-- 3. Info of employees in a particular (unskilled) department
--    (fixed stray "and"; comparisons are case-insensitive by default)
SELECT * FROM U_EMPLOYEE WHERE WORK_AS = 'Carpenter';

-- 4. Total info of a hostel
SELECT * FROM HOSTEL WHERE HOSTEL_NO = 1;

-- 5. Info about the supervisor(s) of a hostel (fixed stray "and")
SELECT F_NAME, L_NAME, SALARY FROM S_EMPLOYEE WHERE POST = 'Supervisor';

-- 6. Salary info of a particular skilled employee's post
SELECT SALARY FROM S_EMPLOYEE WHERE POST = 'Supervisor';

-- 7. Check if an employee was present on a particular day
--    (column is ATTENDENCE, not attendance; fixed quotes/semicolon)
SELECT ATTENDENCE FROM S_EMPLOYEE_TRAN WHERE ID_NO = 'A-1';

-- 8. Number of working days in a month for an employee
SELECT * FROM U_EMPLOYEE;
SELECT WORKING_DAYS FROM U_EMPLOYEE_TRAN WHERE ID_NO = 'A-6';

-- 9. Mess menu for a particular hostel
SELECT F_MESS_id FROM HOSTEL;
SELECT MENU FROM MESS_MENU WHERE MESS_ID = 'M-1';

-- 10. Mess admin info for a particular hostel
SELECT F_MESS_id FROM HOSTEL;
SELECT CATARER, REPRESENTATIVE FROM MESS_ADMIN WHERE MESS_ID = 'M-2';

-- 11. Visitor info for a particular hostel
SELECT * FROM VISITOR WHERE HOSTEL_NO = 2;

-- 12. Arrival/departure time of a student
--    (columns are STUDENT_ARR / STUDENT_DEP, not STUDENT_IN / STUDENT_OUT)
SELECT STUDENT_ARR, STUDENT_DEP FROM STUDENT_IN_OUT WHERE ROLL_NO = 'IPG2013109';

-- 13. Guard present at a particular hostel
SELECT ID_NO, F_NAME, L_NAME, PH_NO FROM U_EMPLOYEE WHERE WORK_AS = 'H_Security';
SELECT * FROM U_EMPLOYEE_TRAN WHERE ATTENDENCE = 'P' AND ID_NO = 'A-6';

-- 14. Total capacity of a particular hostel

-- 15. Whether a student is present/on leave on a given day
--    (column is ATTENDENCE, table STUDENT_TRAN)
